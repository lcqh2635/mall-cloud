# 将本地新创建的仓库、或者已经存在的仓库推送到 Github 远程仓库流程步骤
# 1、在 Github 新建一个空仓库，仓库填写项目名称和描述即可，不要勾选任何 README、.gitignore、license 内容
# 2、在本地执行以下流程
# ✅ 进入项目目录
cd mall-cloud
# ✅ 先初始化，再执行流程
git init
# ✅ 关联 Github 远程仓库
git remote add origin git@github.com:lcqh2635/mall-cloud.git
# ✅ 重命名分支为 main，确保分支名为 main 和 Github 远程主仓库分支名保持一致
git branch -M main
# ✅ 添加所有文件到暂存区
git add .
# ✅ 创建第一次提交，main 分支正式建立
git commit -m "feat: init mall-cloud project structure"
# ✅ 推送并建立追踪关系，-u 让后续直接用 git push 即可
git push -u origin main

# 强制推送命令
git push -u origin main -f
# 基于 SSH 认证克隆仓库
git clone git@github.com:lcqh2635/mall-cloud.git
