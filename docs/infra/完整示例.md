### 📄 4. `db/compose.yaml`（核心数据层）
```yaml
# ==========================================================
# 核心数据层编排文件
# 包含：PostgreSQL（Nacos 专属数据源） + Redis（缓存层）
# ==========================================================

services:
  # ---------------- PostgreSQL ----------------
  postgres:
    image: ${PG_IMAGE}  # 引用 .env 中定义的 PostgreSQL 镜像版本
    container_name: infra-postgres  # 固定容器名称，便于脚本调用与日志追踪
    environment:  # 注入数据库初始化环境变量
      POSTGRES_USER: ${PG_USER}  # 设置数据库超级管理员用户名
      POSTGRES_PASSWORD: ${PG_PASSWORD}  # 设置数据库管理员密码
      POSTGRES_DB: ${PG_DB}  # 首次启动自动创建的专用业务库
      TZ: Asia/Shanghai  # 设置容器时区为东八区，保障事务时间戳一致
    ports:  # 端口映射配置（宿主机:容器）
      - "${PG_PORT}:${PG_PORT}"  # 暴露标准 SQL 端口供外部客户端直连调试
    volumes:  # 数据卷挂载配置
      - pg_data:/var/lib/postgresql/data  # 映射至 PG 默认数据目录，实现 WAL 与表空间持久化
    restart: unless-stopped  # 退出策略：非手动停止则异常退出后自动重启
    healthcheck:  # 健康检查机制配置
      test: ["CMD-SHELL", "pg_isready -U ${PG_USER} -d ${PG_DB}"]  # 调用官方工具验证实例连接状态
      interval: 10s  # 健康检查执行间隔时间
      timeout: 5s  # 单次检查命令最大超时时间
      retries: 3  # 连续失败次数阈值，达标后标记为 unhealthy
      start_period: 20s  # 启动缓冲期，避开 PG 初始化阶段误判
    networks:  # 网络归属声明
      - infra-net  # 加入共享内部网络

  # ---------------- Redis ----------------
  redis:
    image: ${REDIS_IMAGE}  # 引用 .env 中定义的 Redis 镜像版本
    container_name: infra-redis  # 固定容器名称
    command: >  # 覆盖默认启动命令，追加安全与性能参数
      redis-server
      --requirepass ${REDIS_PASSWORD}  # 启用密码认证，拦截未授权访问
      --appendonly yes  # 开启 AOF 持久化，保障写入数据不丢失
      --maxmemory 256mb  # 限制最大内存占用，防止缓存击穿拖垮宿主机
      --maxmemory-policy allkeys-lru  # 内存满载时采用 LRU 算法淘汰最近最少使用键
    ports:  # 端口映射配置
      - "${REDIS_PORT}:${REDIS_PORT}"  # 暴露标准端口供应用层连接
    volumes:  # 数据卷挂载配置
      - redis_data:/data  # 映射至 Redis 工作目录，持久化 RDB/AOF 文件
    restart: unless-stopped  # 退出策略
    healthcheck:  # 健康检查机制配置
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD}", "ping"]  # 携带密码发送 PING 验证服务响应
      interval: 10s  # 检查间隔
      timeout: 5s  # 检查超时
      retries: 3  # 重试次数
      start_period: 10s  # 启动缓冲期
    networks:  # 网络归属声明
      - infra-net  # 加入共享内部网络

# 声明引用的基础网络与卷（Compose 合并运行时自动复用同名定义）
networks:
  infra-net:  # 引用外部定义的网络
    external: false  # 标记为当前项目内部管理，由 00-networks.yaml 提供定义
volumes:
  pg_data:  # 引用外部定义的卷
    external: false
  redis_data:
    external: false
```

```yaml
# ==========================================================
# Nacos 注册与配置中心编排文件
# 说明：已配置直连 PostgreSQL，替代默认内嵌 Derby 数据库
# ==========================================================

services:
  nacos:
    image: ${NACOS_IMAGE}  # 引用 Nacos 官方镜像
    container_name: alibaba-nacos  # 容器命名

      MODE: standalone  # 单机运行模式（开发/测试推荐）

      PGSQL_SERVICE_PORT: ${PG_PORT}  # 指定 PG 服务监听端口
      PGSQL_SERVICE_DB_NAME: ${PG_DB}  # 指定连接的目标数据库名
      PGSQL_SERVICE_USER: ${PG_USER}  # 指定数据库连接账号
      PGSQL_SERVICE_PASSWORD: ${PG_PASSWORD}  # 指定数据库连接密码
      JVM_XMS: ${JVM_XMS_NACOS}  # JVM 初始堆内存限制
      JVM_XMX: ${JVM_XMX_NACOS}  # JVM 最大堆内存限制
      NACOS_AUTH_ENABLE: "true"  # 开启控制台与 OpenAPI 鉴权
      NACOS_AUTH_TOKEN: ${NACOS_TOKEN}  # 鉴权加密盐值配置
    ports:  # 端口暴露配置
      - "${NACOS_HTTP}:${NACOS_HTTP}"  # HTTP 控制台与 RESTful API 端口
      - "${NACOS_GRPC_CLIENT}:${NACOS_GRPC_CLIENT}"  # gRPC 客户端实例注册与配置拉取端口
      - "${NACOS_GRPC_SERVER}:${NACOS_GRPC_SERVER}"  # gRPC 集群节点间数据同步端口
    volumes:  # 持久化挂载配置
      - nacos_data:/home/nacos/data  # 保留配置快照、操作日志与本地运行缓存
    restart: unless-stopped  # 退出策略
    healthcheck:  # 就绪探测配置
      test: ["CMD-SHELL", "curl -f http://localhost:${NACOS_HTTP}/nacos/ || exit 1"]  # HTTP 状态码验证控制台可达性
      interval: 15s  # 检查间隔
      timeout: 10s  # 超时阈值
      retries: 3  # 失败阈值
      start_period: 45s  # 预留 JVM 预热与 PG 连接池初始化时间
    networks:  # 网络配置
      - infra-net  # 加入共享内部网络

networks:
  infra-net:
    external: false
volumes:
  nacos_data:
    external: false
```

### 📄 6. `middleware/rocketmq.yaml`（消息队列）
```yaml
# ==========================================================
# RocketMQ 消息队列编排文件
# 包含：NameServer（路由中心） + Broker（存储与代理）
# ==========================================================

services:
  # ---------------- NameServer ----------------
  rocketmq-namesrv:
    image: ${RMQ_IMAGE}  # 引用统一 RocketMQ 镜像
    container_name: rocketmq-namesrv  # 容器命名
    command: sh mqnamesrv  # 覆盖默认入口命令，仅启动轻量级路由服务
    environment:  # JVM 资源限制配置
      JAVA_OPT_EXT: "-server -Xms${JVM_XMS_RMQ} -Xmx${JVM_XMX_RMQ}"  # 控制路由服务内存水位
    ports:  # 端口暴露配置
      - "${RMQ_NAMESRV}:${RMQ_NAMESRV}"  # 路由注册与元数据发现端口
    volumes:  # 日志持久化挂载
      - rmq_namesrv_logs:/home/rocketmq/logs  # 持久化路由变更与节点心跳日志
    restart: unless-stopped  # 退出策略
    healthcheck:  # 连通性探测配置
      test: ["CMD-SHELL", "nc -z localhost ${RMQ_NAMESRV} || exit 1"]  # TCP 端口监听状态检查
      interval: 10s  # 检查间隔
      timeout: 5s  # 检查超时
      retries: 3  # 重试次数
      start_period: 15s  # 启动缓冲期
    networks:  # 网络归属
      - infra-net  # 加入共享网络

  # ---------------- Broker ----------------
  rocketmq-broker:
    image: ${RMQ_IMAGE}  # 版本与 NameServer 严格一致保障协议兼容
    container_name: rocketmq-broker  # 容器命名
    command: >  # 复合启动命令，指定路由地址、配置文件并开启 5.x 网关
      sh mqbroker -n rocketmq-namesrv:${RMQ_NAMESRV}
      -c /home/rocketmq/rocketmq-5.1.4/conf/broker.conf
      --enable-proxy
    environment:  # JVM 资源限制配置
      JAVA_OPT_EXT: "-server -Xms${JVM_XMS_RMQ} -Xmx${JVM_XMX_RMQ}"  # 控制消息存储内存水位
    ports:  # 多协议端口暴露配置
      - "10909:10909"  # VIP 高速通信通道端口
      - "${RMQ_BROKER}:${RMQ_BROKER}"  # 常规生产者/消费者消息收发端口
      - "10912:10912"  # 主从 CommitLog 异步同步端口
      - "8081:8081"  # Proxy 网关 HTTP/gRPC 端口
    depends_on:  # 启动依赖控制声明
      rocketmq-namesrv:  # 依赖对象名称
        condition: service_healthy  # 等待 NameServer 健康检查通过后再拉起
    volumes:  # 数据与日志持久化挂载
      - rmq_broker_logs:/home/rocketmq/logs  # 保留事务轨迹、堆积报警与消费偏移量日志
      - rmq_broker_store:/home/rocketmq/store  # 物理存储 CommitLog/ConsumeQueue 索引文件
    restart: unless-stopped  # 退出策略
    healthcheck:  # 就绪探测配置
      test: ["CMD-SHELL", "nc -z localhost ${RMQ_BROKER} || exit 1"]  # 主通信端口连通性验证
      interval: 15s  # 检查间隔
      timeout: 5s  # 检查超时
      retries: 3  # 重试次数
      start_period: 35s  # 缓冲期（适配索引加载与内存映射文件初始化）
    networks:  # 网络归属
      - infra-net  # 加入共享网络

networks:
  infra-net:
    external: false
volumes:
  rmq_namesrv_logs:
    external: false
  rmq_broker_logs:
    external: false
  rmq_broker_store:
    external: false
```

### 📄 9. `orchestrate.sh`（统一编排脚本）
```bash
#!/usr/bin/env bash
# ==========================================================
# 中间件集群统一编排启动脚本
# 说明：按依赖顺序拉起各子 compose 文件，结合健康检查保障服务就绪
# ==========================================================

# 启用严格模式：遇错立即退出、未定义变量报错、管道失败传递错误码
set -euo pipefail

# 加载根目录 .env 环境变量（若存在则自动导出至当前 Shell 环境）
if [[ -f .env ]]; then
  set -a  # 开启自动导出所有变量至环境
  source .env
  set +a  # 关闭自动导出
fi

# 定义基础命令与路径变量
COMPOSE_CMD="docker compose"  # Docker Compose 命令别名（兼容 V2 规范）
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"  # 获取脚本所在绝对路径
PROJECT_NAME="infra"  # 统一项目前缀，确保所有文件共享同一网络与卷上下文

echo "🚀 开始按序启动微服务中间件集群..."

# ---------------- 阶段 1：启动核心数据层（PG + Redis） ----------------
echo "📦 [1/4] 启动数据层..."
$COMPOSE_CMD -p $PROJECT_NAME -f "$SCRIPT_DIR/00-networks.yaml" -f "$SCRIPT_DIR/01-volumes.yaml" -f "$SCRIPT_DIR/db/compose.yaml" up -d
# 等待 PG 与 Redis 健康检查通过（需 Docker Compose V2.20+）
$COMPOSE_CMD -p $PROJECT_NAME -f "$SCRIPT_DIR/db/compose.yaml" wait postgres --timeout=60
$COMPOSE_CMD -p $PROJECT_NAME -f "$SCRIPT_DIR/db/compose.yaml" wait redis --timeout=30
echo "✅ 数据层已就绪"

# ---------------- 阶段 2：启动 Nacos（强依赖 PG 连接） ----------------
echo "📦 [2/4] 启动 Nacos 注册配置中心..."
```

### 🛡️ 核心使用与避坑指南

1. **赋予脚本执行权限并运行**
   ```bash
   chmod +x orchestrate.sh
   ./orchestrate.sh
   ```

2. **Nacos 首次启动必做：初始化 PostgreSQL 表结构**  
   Nacos 对接外部数据库时**不会自动建表**。启动后需手动导入官方 PG 脚本：
   ```bash
   # 1. 下载 nacos-pg.sql（从 Nacos GitHub Release 或官方文档获取）
   # 2. 导入至 PG 容器
   docker exec -i infra-postgres psql -U nacos_admin -d nacos_config < nacos-pg.sql
   # 3. 重启 Nacos 使其加载元数据表
   docker compose -p infra -f middleware/nacos.yaml restart
   ```

3. **跨文件依赖原理说明**  
   脚本中每次调用均携带 `-p infra` 参数，Docker Compose 会将所有文件视为**同一个项目上下文**。因此 `00-networks.yaml` 定义的网络会自动被后续文件复用，无需 `external: true`，彻底避免 `network not found` 报错。

4. **生产环境升级建议**
    - 🔒 **密码脱敏**：将 `.env` 加入 `.gitignore`，配合 CI/CD 变量注入或 `docker secrets`
    - 🌐 **端口收敛**：若应用与中间件同机部署，注释所有 `ports` 映射，仅靠 `infra-net` 内部通信
    - 💾 **Seata 存储切换**：将 `STORE_MODE: db`，并在 Nacos 配置中心下发 `store.db.url` 指向高可用数据库（可复用本 PG 或独立部署）
